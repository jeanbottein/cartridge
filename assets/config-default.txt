# •cartridge settings. The cartridge's folders are fixed, next to system/:
#   games/          one folder per game
#   mods/           patches and file replacements
#   saves-copies/   backups of your save files
# Need one elsewhere (another drive...)? Replace it with a symbolic link.

# ==========================================
# 1. GENERAL SETTINGS
# ==========================================

# Background music (optional)
# Plays the audio files found under this system/ folder (mp3, flac, ogg,
# wav, m4a/aac), shuffled and looping; fades out while a game runs.
# Set to False to keep the UI silent; Select on a gamepad (M on a keyboard)
# toggles this from anywhere in the UI.
#MUSIC_ENABLED=True

# Background music volume, in percent (0-100).
#MUSIC_VOLUME=50

# Kiosk mode: children-proof, only the Library and only launching games
# (no editing, no Steam button, no other tab, no quitting).
# To leave it: hold L1 + R1 + X on a gamepad (Q + E + X on a keyboard) for
# 5 seconds, or set this back to False here.
#KIOSK_MODE=False

# Widescreen layout: auto, on or off
# auto: side panels on screens 3:2 or wider (16:10, 16:9...), 4:3 otherwise;
# portrait screens get the 4:3 layout extended downwards.
# Default: auto
#WIDESCREEN=auto

# Look of the interface: midnight or retro
# (colors, font and shapes; the layout stays the same)
# Default: midnight
#THEME=midnight

# Window mode: borderless or window
# borderless: fullscreen at the display's native resolution
# window: a window you can resize (under gamescope it fills the screen anyway)
# Default: borderless
#WINDOW_MODE=borderless


# ==========================================
# 2. GAME LAUNCHERS & INTEGRATION
# ==========================================

# Games in Steam: choose per game in the Library (Y), remembered in
# system/steam.json. Shortcuts are tagged with this cartridge's id
# (system/cartridge-id), so your other non-Steam games, and other cartridges'
# games, are never touched. Steam shows changes after it restarts.

# Optionally target a specific Steam user ID (leave commented to update all users)
#STEAM_USERID=

# Windows games run with Steam's Proton: the build their game.json names
# ("proton"), else Steam's default, else the newest one installed.
# Keep what winetricks downloads for them in system/winetricks, so this
# cartridge brings it to other machines (always used when it is there)
#PROTON_KEEP_DOWNLOADS=False

# Add this cartridge itself to Steam under its folder name (optional), with retro
# artwork, or your own: cover.png, wide.png, hero.png, logo.png, icon.png in system/
# The shortcut launches this app.
#STEAM_ADD_SELF=False

# If you want to automatically download artwork (grids, heroes, logos) from SteamGridDB,
# you must provide a free API key from https://www.steamgriddb.com/profile/preferences/api
#STEAMGRIDDB_API_KEY=

# SteamGridDB content filters (set to True to include)
#STEAMGRIDDB_NSFW=False
#STEAMGRIDDB_HUMOR=False
#STEAMGRIDDB_EPILEPSY=False

# Manifest export (optional): games/manifests.json for Steam ROM Manager
# Set to False to skip writing manifests.json
#MANIFEST_EXPORT=True


# ==========================================
# 3. SHARING ON THE LAN
# ==========================================

# Let other •cartridge machines on the network see and copy your games (optional).
# No password: only enable on a network you trust.
#SHARE_ENABLED=False

# TCP (HTTP) + UDP (discovery) port. If other machines can't see this one, open it:
#   sudo firewall-cmd --permanent --add-port=47800/tcp --add-port=47800/udp && sudo firewall-cmd --reload
#SHARE_PORT=47800

# Machines to list even when the LAN broadcast can't reach them (VPN, other subnet)
# Format: host[:port], comma-separated
#SHARE_PEERS=192.168.1.20,library.local:47800


# ==========================================
# 4. DEBUG
# ==========================================

# Only with a file named "debug" next to the app (debug mode): then SETTINGS
# shows this section, and these keys apply.

# Interface only: hide the Steam buttons and dim the Steam settings, as on a
# machine without Steam.
#DEBUG_NO_STEAM=False

# Interface only: loop a fake library update in the Library's progress card,
# to see it (a real update is often too quick). Nothing runs.
#DEBUG_FAKE_PROGRESS=False
